23: 5. Require original directed relation-role evidence for each replacement donor with the same counterpart general type. 1: """Count directed multirelational edges; each self-loop contributes two total."""
2: 
3: 
4: def compute_degrees(graph):
5:     degrees = {n['id']: {'in_degree': 0, 'out_degree': 0, 'total_degree': 0}
6:                for n in graph['nodes']}
7:     for edge in graph['edges']:
8:         degrees[edge['source']]['out_degree'] += 1
9:         degrees[edge['target']]['in_degree'] += 1
10:     for values in degrees.values():
11:         values['total_degree'] = values['in_degree'] + values['out_degree']
12:     return degrees

147: def random_baseline(original, hub_edges, seed, match_degree=True):
148:     """Random eligible replacement, exact source-edge/strata matching.
149: 
150:     Try shuffled same-degree-bin donors first. Fall back to shuffled compatible
151:     eligible donors when no legal same-bin replacement exists. Trials reject
152:     original/QA-positive triples, duplicates, loops, identity and wrong position.
153:     """
154:     nodes = {n['id']: n for n in original['nodes']}
155:     deg = degrees(nodes, original['edges'])
156:     role = defaultdict(set)
157:     for e in original['edges']:
158:         s, r, t = triple(e)
159:         role[('source', r, nodes[t]['entity_type'])].add(s)
160:         role[('target', r, nodes[s]['entity_type'])].add(t)
161:     forbidden = {triple(e) for e in original['edges']} | qa_positive_triples(original)
162:     rng, seen, result = random.Random(seed), set(), []
163:     diagnostics = Counter()
164:     for hub in hub_edges:
165:         provenance = hub['injection_provenance']
166:         positions = [s['position'] for s in provenance['swaps']]
167:         if len(positions) != 1:
168:             raise ValueError('Baseline supports exactly one changed endpoint; refusing unmatched strata.')
169:         position = positions[0]
170:         base = original['edges'][provenance['source_edge_index']]
171:         old = base[position]
172:         otherpos = 'target' if position == 'source' else 'source'
173:         pool = sorted(q for q in role[(position, base['relation'], nodes[base[otherpos]]['entity_type'])]
174:                       if is_eligible(nodes[q]) and nodes[q]['entity_type'] == nodes[old]['entity_type'] and q != old)
175:         preferred, fallback = [], []
176:         desired = degree_bin(deg[hub[position]]['total'])

186:                 mapping = {old: q, q: old}
187:                 s, r, t = mapping.get(base['source'], base['source']), base['relation'], mapping.get(base['target'], base['target'])
