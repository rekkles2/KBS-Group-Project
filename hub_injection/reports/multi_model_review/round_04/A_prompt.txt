        writer=csv.DictWriter(stream,fieldnames=edge_fields);writer.writeheader()

            if synthetic:
                synthetic_count+=1;outgoing[edge['source']]+=1;incoming[edge['target']]+=1
            writer.writerow({'edge_index':index,'source':edge['source'],'relation':edge['relation'],'target':edge['target'],
                'is_synthetic_edge':int(synthetic),'synthetic_class':'fake' if synthetic else 'original',
                'factual_verification_status':edge['factual_verification']['status'] if synthetic else 'not_adjudicated'})
    node_fields=['qid','entity_type','is_synthetic_node','incident_to_synthetic_edge','synthetic_in_degree','synthetic_out_degree','synthetic_total_degree']
    incident=0
    with (outputs/'node_injection_labels.csv').open('w',encoding='utf-8',newline='') as stream:
        writer=csv.DictWriter(stream,fieldnames=node_fields);writer.writeheader()
        for node in sorted(augmented['nodes'],key=lambda n:n['id']):
            qid=node['id'];total=incoming[qid]+outgoing[qid];incident+=int(total>0)
            writer.writerow({'qid':qid,'entity_type':node['entity_type'],'is_synthetic_node':int(qid not in original_nodes),

                'synthetic_out_degree':outgoing[qid],'synthetic_total_degree':total})
    return {'edge_rows':len(augmented['edges']),'synthetic_edges':synthetic_count,'node_rows':len(augmented['nodes']),

import copy
import random
from collections import defaultdict, Counter
from decimal import Decimal, ROUND_HALF_EVEN
from .load_data import triple

SYNTHETIC_SEMANTICS = ('Synthetic overlay visibility inherited from a source QA-projection edge; '
                       'not QA support, historical truth, or a Wikidata validity interval.')


def injection_budget(original_unique_edges, ratio):
    ratio = Decimal(str(ratio))
    if not ratio.is_finite() or ratio < 0:
        raise ValueError('injection-ratio must be finite and nonnegative')
    return int((Decimal(original_unique_edges) * ratio).quantize(Decimal('1'), rounding=ROUND_HALF_EVEN))


def balanced_sample(candidates, count, seed):
    """Round-robin types, then relations within each type; seeded bucket order.

    Deduplication occurs before sampling. Uniform marginals are limited by
    available candidates; this is not sampling proportional to original edges.
    """
    rng = random.Random(seed)
    buckets = defaultdict(lambda: defaultdict(list))
    for candidate in candidates:
        buckets[candidate['entity_type']][candidate['relation']].append(candidate)
    types = sorted(buckets)
    rng.shuffle(types)
    relations, cursors = {}, Counter()
    for kind in types:
        relations[kind] = sorted(buckets[kind])
        rng.shuffle(relations[kind])
        for relation in relations[kind]:
            # Canonical caller-independent order before the seeded permutation.
            buckets[kind][relation].sort(key=triple)
            rng.shuffle(buckets[kind][relation])
    result = []
    while len(result) < count:
        progress = False
        for kind in types:
            available = relations[kind]
            for _ in range(len(available)):
                relation = available[cursors[kind] % len(available)]
                cursors[kind] += 1
                if buckets[kind][relation]:
                    result.append(buckets[kind][relation].pop())
                    progress = True
                    break
            if len(result) == count:
                break
        if not progress:
            break
    return sorted(result, key=triple)


def make_injected_edges(graph, chosen):
