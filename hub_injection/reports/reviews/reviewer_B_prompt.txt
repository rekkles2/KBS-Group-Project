## Method

1. Audit all input objects, CSV agreement, snapshots, evidence and input SHA-256 hashes.
2. On original unique directed multirelational edges, total degree = incoming + outgoing (self-loop contributes two). Rank independently per `entity_type`, breaking ties by lexical QID. Require true eligibility, no flags, valid type and positive degree.
3. Initially choose `min(n,20,max(2,ceil(.05*n)))` per type with at least two eligible nodes. `--top-k-max` is the initial cap. Double ranked prefixes deterministically only if the candidate pool is insufficient; constraints remain unchanged.
4. Enumerate pair-local transpositions of hub pairs. For each original incident edge apply the same pair mapping to both endpoints. Many candidate-specific mappings are possible; the final overlay is **not a single global graph permutation**. This is the requested additive pairwise adaptation.
5. Require original directed relation-role evidence for each replacement donor with the same counterpart general type. This checks an empirical domain/range signature, not a formal ontology or factual truth. Exclude self-loops, all original triples across snapshots, conservatively projectable positive triples from all QA, and duplicate candidates.
6. Choose exactly `round_half_even(Decimal(original_edge_count)*Decimal(ratio))`. The default is 774 additional edges (10.003877% of 7,737 due to integer rounding), yielding 8,511 edges. Sample with seed 42 using round-robin types and relations, seeded within buckets. This balance differs from original-distribution-proportional sampling.
7. Preserve all original objects unchanged. Append explicitly synthetic edges with empty `evidence`, pair/source-edge/swap provenance, and original QA references only under `injection_provenance`. Original snapshot lists remain annotation observations; injected lists indicate synthetic overlay visibility. Original node observation lists remain unchanged even when an overlay endpoint has no original observation in that snapshot.

Missing edges are not necessarily false. All injected factual statuses are `unverified`; zero have been independently established false. Queries and answers are preserved, and no attack-effectiveness claim is made.

## Relationship to prior work

[LogicPoison](https://arxiv.org/html/2604.02954), Sections 4.1–4.3, ranks within-type document frequencies, combines global hubs and query-specific entities, then rewrites corpus mentions via a cyclic bijection. This project operates on an existing QA-projection graph, ranks explicit graph degree, uses pairwise swaps, and appends a budgeted overlay without rewriting source text or removing edges. It is an adaptation, not an exact reproduction. [PAT-Questions](https://arxiv.org/html/2402.11034v2), Sections 3–4, generates present-anchored QA using Wikidata/SPARQL. Local annotation-observation snapshots do not reconstruct statement qualifiers or validity intervals.

"""Count directed multirelational edges; each self-loop contributes two total."""


def compute_degrees(graph):
    degrees = {n['id']: {'in_degree': 0, 'out_degree': 0, 'total_degree': 0}
               for n in graph['nodes']}
    for edge in graph['edges']:
        degrees[edge['source']]['out_degree'] += 1
        degrees[edge['target']]['in_degree'] += 1
    for values in degrees.values():
        values['total_degree'] = values['in_degree'] + values['out_degree']
    return degrees

"""Pairwise candidate enumeration contributed by OpenCode kimi-k2.6.

added duplicate/isolated-hub guards and validated original edge references.

import itertools
from collections import defaultdict
from .select_hubs import is_eligible


def generate_candidates(graph, hubs, forbidden_triples=None):
    node_map = {n['id']: n for n in graph['nodes']}
    edges = graph['edges']
    relations = {r['id'] for r in graph['relations']}
    validated = {}
    for kind, qids in hubs.items():
        if len(set(qids)) != len(qids):
            raise ValueError(f'Duplicate hub identifiers for {kind}')
        for qid in qids:
            if qid not in node_map:
                raise ValueError(f'Hub {qid} missing')
            node = node_map[qid]
            if node['entity_type'] != kind or not is_eligible(node):
                raise ValueError(f'Hub {qid} has mismatched type, flags or ineligibility')
        validated[kind] = list(qids)

    incidence = defaultdict(set)
    out_evidence, in_evidence = defaultdict(set), defaultdict(set)
    original_triples = set()
    for index, edge in enumerate(edges):
        source, relation, target = edge['source'], edge['relation'], edge['target']
        if source not in node_map or target not in node_map or relation not in relations:
            raise ValueError(f'Invalid original edge reference at index {index}')
        incidence[source].add(index)
        incidence[target].add(index)
        source_type, target_type = node_map[source]['entity_type'], node_map[target]['entity_type']
        out_evidence[source].add((relation, target_type))
        in_evidence[target].add((relation, source_type))
        original_triples.add((source, relation, target))
    for qids in validated.values():
        for qid in qids:
            if not incidence[qid]:
                raise ValueError(f'Isolated hub {qid}')

    forbidden_set = set(forbidden_triples or ())
    candidates, seen, rejected, attempted = [], set(), defaultdict(int), 0
    for kind in sorted(validated):
        for a, b in itertools.combinations(sorted(validated[kind]), 2):
            for edge_index in sorted(incidence[a] | incidence[b]):
                attempted += 1
                edge = edges[edge_index]
                old_source, relation, old_target = edge['source'], edge['relation'], edge['target']
                source = b if old_source == a else (a if old_source == b else old_source)
                target = b if old_target == a else (a if old_target == b else old_target)
                changed = [position for position, old, new in
                           [('source',old_source,source),('target',old_target,target)] if old != new]
                proposed = source, relation, target
                reason = None
                if source == target:
                    reason = 'self_loop'
                elif source not in node_map or target not in node_map:
                    reason = 'invalid_endpoint'
                elif 'source' in changed and (relation,node_map[target]['entity_type']) not in out_evidence[source]:
                    reason = 'missing_source_donor_evidence'
                elif 'target' in changed and (relation,node_map[source]['entity_type']) not in in_evidence[target]:
                    reason = 'missing_target_donor_evidence'
                elif proposed in original_triples:
                    reason = 'original_triple'
                elif proposed in forbidden_set:
                    reason = 'forbidden_triple'
                elif proposed in seen:
                    reason = 'duplicate'
                if reason:
                    rejected[reason] += 1
                    continue
                seen.add(proposed)
                candidates.append({'source':source, 'relation':relation, 'target':target,
                    'source_edge_index':edge_index, 'hub_pair':[a,b], 'entity_type':kind,
                    'changed_positions':changed})
    candidates.sort(key=lambda c:(c['source'],c['relation'],c['target'],c['hub_pair'][0],c['hub_pair'][1]))
    return candidates, {'attempted':attempted, 'accepted_unique':len(candidates),
                        'rejected_by_reason':dict(sorted(rejected.items()))}

"""Balanced deterministic sampling and explicitly synthetic provenance."""
import copy
import random
from collections import defaultdict, Counter
from decimal import Decimal, ROUND_HALF_EVEN
from .load_data import triple

SYNTHETIC_SEMANTICS = ('Synthetic overlay visibility inherited from a source QA-projection edge; '
                       'not QA support, historical truth, or a Wikidata validity interval.')


def injection_budget(original_unique_edges, ratio):
    ratio = Decimal(str(ratio))
    if not ratio.is_finite() or ratio < 0:
        raise ValueError('injection-ratio must be finite and nonnegative')
    return int((Decimal(original_unique_edges) * ratio).quantize(Decimal('1'), rounding=ROUND_HALF_EVEN))


def balanced_sample(candidates, count, seed):
    """Round-robin types, then relations within each type; seeded bucket order.

    Deduplication occurs before sampling. Uniform marginals are limited by
    available candidates; this is not sampling proportional to original edges.
    """
    rng = random.Random(seed)
    buckets = defaultdict(lambda: defaultdict(list))
    for candidate in candidates:
        buckets[candidate['entity_type']][candidate['relation']].append(candidate)
    types = sorted(buckets)
    rng.shuffle(types)
    relations, cursors = {}, Counter()
    for kind in types:
        relations[kind] = sorted(buckets[kind])
        rng.shuffle(relations[kind])
        for relation in relations[kind]:
            buckets[kind][relation].sort(key=triple)
            rng.shuffle(buckets[kind][relation])
    result = []
    while len(result) < count:
        progress = False
        for kind in types:
            available = relations[kind]
            for _ in range(len(available)):
                relation = available[cursors[kind] % len(available)]
                cursors[kind] += 1
                if buckets[kind][relation]:
                    result.append(buckets[kind][relation].pop())
                    progress = True
                    break
            if len(result) == count:
                break
        if not progress:
            break
    return sorted(result, key=triple)


def make_injected_edges(graph, chosen):
    result = []
    for candidate in chosen:
        original = graph['edges'][candidate['source_edge_index']]
        pair = candidate['hub_pair']
        swaps = [{'position': position, 'from': original[position], 'to': candidate[position]}
                 for position in ('source', 'target') if candidate[position] != original[position]]
        result.append({
            'source': candidate['source'], 'relation': candidate['relation'], 'target': candidate['target'],
            'observed_in_snapshots': list(original['observed_in_snapshots']), 'evidence': [],
            'edge_origin': 'synthetic_injection', 'is_poisoned': True,
            'injection_method': 'type_preserving_hub_swap',
            'injection_snapshots': list(original['observed_in_snapshots']),
            'source_observed_snapshots': list(original['observed_in_snapshots']),
            'snapshot_semantics': SYNTHETIC_SEMANTICS,
            'injection_provenance': {
                'source_edge_index': candidate['source_edge_index'],
                'original_source': original['source'], 'original_relation': original['relation'],
                'original_target': original['target'], 'hub_pair': pair,
                'pair_permutation': {pair[0]: pair[1], pair[1]: pair[0]},
                'entity_type': candidate['entity_type'], 'swaps': swaps,
                'source_observed_snapshots': list(original['observed_in_snapshots']),
                'source_question_ids': sorted({e['question_id'] for e in original['evidence']}),
                'source_evidence_references': copy.deepcopy(original['evidence']),
                'source_evidence_semantics': 'Source-edge derivation only; does not support the injected claim.',
                'compatibility_policy': 'original_directed_relation_role_and_counterpart_general_type',
                'original_qa_positive_evidence': 'no_identical_conservatively_projectable_triple',
                'factual_verification_status': 'unverified'
            }
        })
    return result


def augment_graph(graph, injected, manifest):
    # Share read-only original objects; no caller function may mutate them.
    result = dict(graph)
    result['edges'] = list(graph['edges']) + injected
    result['metadata'] = dict(graph['metadata'])
    if 'task1_hub_injection' in result['metadata']:
        raise ValueError('Input already has Task 1 injection metadata')
    result['metadata']['task1_hub_injection'] = {
        'method': 'additive_type_preserving_pairwise_hub_swap',
        'source_statistics_semantics': 'metadata.stats is unchanged original-source statistics.',
        'snapshot_semantics': SYNTHETIC_SEMANTICS,
        'node_visibility_policy': 'Original node observation lists remain unchanged; overlay endpoints do not assert node QA occurrence.',
        'augmented_statistics': {'nodes': len(graph['nodes']), 'edges': len(result['edges']),
                                 'injected_edges': len(injected), 'qa_records': len(graph['questions'])},
        'configuration': manifest['configuration'], 'input_sha256': manifest['input_sha256'],
        'independently_verified_false_count': 0, 'unverified_synthetic_count': len(injected)
    }
    return result

"""Read input objects without mutating or duplicating source files."""
import csv
import hashlib
import json
from pathlib import Path


def sha256(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(chunk)
    return digest.hexdigest()


def load_data(graph_path, csv_path):
    with Path(graph_path).open(encoding='utf-8-sig') as stream:
        graph = json.load(stream)
    with Path(csv_path).open(encoding='utf-8-sig', newline='') as stream:
        rows = list(csv.DictReader(stream))
    return graph, rows, {'graph': sha256(graph_path), 'entity_csv': sha256(csv_path)}


def write_json(path, obj):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(obj, ensure_ascii=False, indent=2, sort_keys=True) + '\n', encoding='utf-8')


def triple(edge):
    return edge['source'], edge['relation'], edge['target']

def comparison_table(original, augmented, field):
    lines = ['| Category | Original | Augmented |', '|---|---:|---:|']
    for key in sorted(set(original[field]) | set(augmented[field])):
        lines.append(f'| {key} | {original[field].get(key, 0)} | {augmented[field].get(key, 0)} |')
    return '\n'.join(lines)

"""Rank using the ORIGINAL graph only; deterministic lexical QID ties."""
import math
from collections import defaultdict


def is_eligible(node):
    annotation = node.get('general_type_annotation', {})
    return (annotation.get('eligible_for_type_swap') is True
            and isinstance(node.get('entity_type'), str)
            and node['entity_type'].strip() not in {'', 'Unknown', 'unknown', 'Unresolved', 'Invalid'}
            and not annotation.get('data_quality_flags')
            and not node.get('data_quality_flags'))


def rank_eligible(graph, degrees):
    groups = defaultdict(list)
    for node in graph['nodes']:
        if is_eligible(node) and degrees[node['id']]['total_degree'] > 0:
            groups[node['entity_type']].append(node['id'])
    return {kind: sorted(ids, key=lambda q: (-degrees[q]['total_degree'], q))
            for kind, ids in sorted(groups.items())}


def initial_k(groups, top_k_max=20):
    if top_k_max < 2:
        raise ValueError('top-k-max must be at least 2')
    return {kind: min(len(ids), top_k_max, max(2, math.ceil(0.05 * len(ids))))
            if len(ids) >= 2 else 0 for kind, ids in groups.items()}


def expand_k(groups, sizes):
    """Deterministically double each selected prefix, up to complete eligibility."""
    return {kind: min(len(groups[kind]), max(k + 1, k * 2)) if k else 0
            for kind, k in sizes.items()}


def selected_hubs(groups, sizes):
    return {kind: ids[:sizes[kind]] for kind, ids in groups.items() if sizes[kind] >= 2}

"""Input integrity and conservative positive-evidence checks over all QA."""
import re
from collections import Counter
from .load_data import triple

QID = re.compile(r'^Q[1-9][0-9]*$')
PID = re.compile(r'^P[1-9][0-9]*$')
EXPECTED = {'nodes': 7868, 'edges': 7737, 'relations': 20, 'questions': 18436, 'unresolved_paths': 496}


def qa_positive_triples(graph):
    """No Cartesian second-hop pairing when more than one intermediate exists.

    Include valid first hops even in excluded QA; include second hops only for
    exactly one intermediate. These are positive annotation evidence, not a
    complete set of historical Wikidata facts.
    """
    positives = set()
    for question in graph['questions']:
        record = question['source_record']
        subject = record.get('subject', {}).get('subject')
        rels = record.get('relations', [])
        answers = [x.get('ID') for x in record.get('answer annotations', [])]
        # Duplicate annotations of the same QID do not create pairing ambiguity.
        mids = list(dict.fromkeys(x.get('ID') for x in record.get('intermediate entities', [])))
        proposed = []
        if len(rels) == 1:
            proposed.extend((subject, rels[0], answer) for answer in answers)
        elif len(rels) == 2:
            proposed.extend((subject, rels[0], mid) for mid in mids)
            if len(mids) == 1:
                proposed.extend((mids[0], rels[1], answer) for answer in answers)
        for s, r, t in proposed:
            if (isinstance(s, str) and QID.fullmatch(s) and isinstance(t, str)
                    and QID.fullmatch(t) and isinstance(r, str) and PID.fullmatch(r)):
                positives.add((s, r, t))
    return positives


def validate_original(graph, rows):
    for key in ('metadata', 'nodes', 'relations', 'edges', 'questions', 'unresolved_paths'):
        if key not in graph:
            raise ValueError(f'Missing top-level field: {key}')
    nodes = {n['id']: n for n in graph['nodes']}
    relations = {r['id'] for r in graph['relations']}
    questions = {q['id']: q for q in graph['questions']}
    snapshots = set(graph['metadata']['snapshots'])
    csv_rows = {r['wikidata_id']: r for r in rows}
    checks = {}
    def check(name, condition):
        checks[name] = bool(condition)
        if not condition:
            raise ValueError(f'Original integrity check failed: {name}')
    check('unique_qids', len(nodes) == len(graph['nodes']))
    check('valid_qids', all(QID.fullmatch(q) for q in nodes))
    check('unique_relations', len(relations) == len(graph['relations']))
    check('valid_relations', all(PID.fullmatch(r) for r in relations))
    check('unique_qa_ids', len(questions) == len(graph['questions']))
    check('unique_original_triples', len({triple(e) for e in graph['edges']}) == len(graph['edges']))
    check('csv_unique_qids', len(csv_rows) == len(rows))
    check('csv_node_coverage', csv_rows.keys() == nodes.keys())
    check('known_snapshot_ids', snapshots == {'Dec2021', 'Dec2023', 'March2024'})
    for node in graph['nodes']:
        row = csv_rows[node['id']]
        ann = node['general_type_annotation']
        check_node = (row['entity_type'] == node['entity_type']
                      and row['eligible_for_type_swap'] == str(ann['eligible_for_type_swap'])
                      and set(filter(None, row['data_quality_flags'].split('|'))) == set(ann['data_quality_flags'])
                      and row['type_policy_version'] == ann['policy_version']
                      and set(filter(None, row['detailed_type_ids'].split('|'))) == {d['id'] for d in node['detailed_entity_types']}
                      and isinstance(ann['eligible_for_type_swap'], bool)
                      and isinstance(ann['data_quality_flags'], list)
                      and isinstance(node['entity_type'], str)
                      and set(node['observed_in_snapshots']) <= snapshots)
        if not check_node:
            raise ValueError(f'Node/CSV disagreement: {node["id"]}')
    checks['all_node_csv_types_eligibility_flags_consistent'] = True
    for edge in graph['edges']:
        if not (edge['source'] in nodes and edge['target'] in nodes and edge['relation'] in relations):
            raise ValueError(f'Dangling original edge: {triple(edge)}')
        if not set(edge['observed_in_snapshots']) <= snapshots:
            raise ValueError('Invalid edge snapshot')
        evidence_snapshots = set()
        for evidence in edge['evidence']:
            if evidence['question_id'] not in questions or evidence['hop'] not in (1, 2):
                raise ValueError('Invalid evidence reference')
            question = questions[evidence['question_id']]
            evidence_snapshots.add(question['snapshot'])
            if (triple(edge) not in qa_positive_triples({'questions': [question]})
                    or question['source_record']['relations'][evidence['hop'] - 1] != edge['relation']):
                raise ValueError('Evidence does not project to cited edge')
        if evidence_snapshots != set(edge['observed_in_snapshots']):
            raise ValueError('Original snapshot/evidence mismatch')
        for endpoint in (edge['source'], edge['target']):
            if not set(edge['observed_in_snapshots']) <= set(nodes[endpoint]['observed_in_snapshots']):
                raise ValueError('Original edge visibility missing from endpoint observations')
    checks['original_endpoints_relations_evidence_snapshots_valid'] = True
    check('qa_snapshot_ids', all(q['snapshot'] in snapshots for q in graph['questions']))
    check('unresolved_qa_references', all(u['question_id'] in questions and u['snapshot'] in snapshots
                                         for u in graph['unresolved_paths']))
    for key in ('snapshot_semantics', 'node_snapshot_semantics', 'projection_semantics', 'pairing_policy'):
        check(f'metadata_{key}', bool(graph['metadata'].get(key)))
    counts = {key: len(graph[key]) for key in EXPECTED}
    return {'checks': checks, 'counts': counts, 'expected_counts': EXPECTED,
            'discrepancies': {k: {'expected': EXPECTED[k], 'actual': v} for k, v in counts.items() if v != EXPECTED[k]},
            'qa_positive_triples': len(qa_positive_triples(graph)),
            'flag_counts': dict(Counter(flag for n in graph['nodes'] for flag in n['general_type_annotation']['data_quality_flags'])),
            'qa_records_with_duplicate_intermediate_qids': sum(
                len(q['source_record'].get('intermediate entities', [])) !=
                len({x.get('ID') for x in q['source_record'].get('intermediate entities', [])})
                for q in graph['questions'])}

"""Independent validation reconstructs transformations and directed evidence."""
from collections import defaultdict
from .load_data import triple
from .select_hubs import is_eligible
from .validate_original import qa_positive_triples
from .inject_misinformation import SYNTHETIC_SEMANTICS


def validate_poisoned(original, poisoned, selected, target, require_target=True):
    checks = {}
    def check(name, condition):
        checks[name] = bool(condition)
        if not condition:
            raise ValueError(f'Poisoned validation failed: {name}')
    for key, value in original.items():
        if key not in {'edges', 'metadata'}:
            check(f'preserved_{key}', poisoned.get(key) == value)
    check('preserved_original_metadata', all(poisoned['metadata'].get(k) == v for k, v in original['metadata'].items()))
    length = len(original['edges'])
    check('preserved_original_edges_in_order', poisoned['edges'][:length] == original['edges'])
    injected = poisoned['edges'][length:]
    check('exact_budget_or_documented_shortfall', len(injected) == target if require_target else len(injected) <= target)
    check('unique_augmented_triples', len({triple(e) for e in poisoned['edges']}) == len(poisoned['edges']))
    namespace = poisoned['metadata']['task1_hub_injection']
    check('accurate_namespaced_augmented_statistics', namespace['augmented_statistics'] == {
        'nodes':len(original['nodes']), 'edges':len(poisoned['edges']),
        'injected_edges':len(injected), 'qa_records':len(original['questions'])})
    check('accurate_namespaced_verification_counts', namespace['independently_verified_false_count'] == 0
          and namespace['unverified_synthetic_count'] == len(injected))
    nodes = {n['id']: n for n in original['nodes']}
    relations = {r['id'] for r in original['relations']}
    positives = qa_positive_triples(original)
    # These signatures are reconstructed here independently of the worker.
    incoming, outgoing = defaultdict(set), defaultdict(set)
    for edge in original['edges']:
        outgoing[edge['source']].add((edge['relation'], nodes[edge['target']]['entity_type']))
        incoming[edge['target']].add((edge['relation'], nodes[edge['source']]['entity_type']))
    for edge in injected:
        check('injected_references_resolve', edge['source'] in nodes and edge['target'] in nodes and edge['relation'] in relations)
        check('no_injected_self_loops', edge['source'] != edge['target'])
        check('no_qa_positive_triples', triple(edge) not in positives)
        check('explicit_synthetic_identity', edge['edge_origin'] == 'synthetic_injection'
              and edge['is_poisoned'] is True and edge['injection_method'] == 'type_preserving_hub_swap')
        provenance = edge['injection_provenance']
        index = provenance['source_edge_index']
        check('valid_source_edge_index', isinstance(index, int) and 0 <= index < length)
        source = original['edges'][index]
        check('exact_source_triple', (provenance['original_source'], provenance['original_relation'], provenance['original_target']) == triple(source))
        a, b = provenance['hub_pair']
        check('eligible_same_type_hubs', a != b and a in nodes and b in nodes
              and is_eligible(nodes[a]) and is_eligible(nodes[b])
              and nodes[a]['entity_type'] == nodes[b]['entity_type'] == provenance['entity_type'])
        check('selected_hubs_only', a in selected[provenance['entity_type']] and b in selected[provenance['entity_type']])
        permutation = {a: b, b: a}
        expected = (permutation.get(source['source'], source['source']), source['relation'],
                    permutation.get(source['target'], source['target']))
        check('pairwise_permutation_both_endpoints', triple(edge) == expected and triple(edge) != triple(source)
              and provenance['pair_permutation'] == permutation)
        swaps = [{'position': p, 'from': source[p], 'to': edge[p]}
                 for p in ('source', 'target') if source[p] != edge[p]]
        check('complete_swap_provenance', provenance['swaps'] == swaps)
        for swap in swaps:
            position, donor = swap['position'], swap['to']
            role_evidence = outgoing[donor] if position == 'source' else incoming[donor]
            counterpart = edge['target'] if position == 'source' else edge['source']
            check('directed_role_compatibility', (edge['relation'], nodes[counterpart]['entity_type']) in role_evidence)
        snaps = source['observed_in_snapshots']
        check('source_overlay_snapshot_identity', edge['observed_in_snapshots'] == edge['injection_snapshots']
              == edge['source_observed_snapshots'] == provenance['source_observed_snapshots'] == snaps)
        check('synthetic_snapshot_semantics', edge['snapshot_semantics'] == SYNTHETIC_SEMANTICS)
        check('qa_evidence_not_synthetic_support', edge['evidence'] == []
              and provenance['source_question_ids'] == sorted({e['question_id'] for e in source['evidence']})
              and provenance['source_evidence_references'] == source['evidence'])
        check('honest_factual_status', provenance['factual_verification_status'] == 'unverified')
        # Exact allowed fields exclude accidental historical validity qualifiers.
        check('no_added_validity_fields', set(edge) == {
            'source','relation','target','observed_in_snapshots','evidence','edge_origin','is_poisoned',
            'injection_method','injection_snapshots','source_observed_snapshots','snapshot_semantics','injection_provenance'})
    return {'passed': all(checks.values()), 'checks': checks, 'injected_count': len(injected),
            'total_edges': len(poisoned['edges'])}

"""Execute unittest and save actual machine-readable and textual results."""
import io
import json
from pathlib import Path
import sys
import unittest

ROOT = Path(__file__).resolve().parent


def main():
    suite = unittest.defaultTestLoader.discover(str(ROOT / 'tests'), top_level_dir=str(ROOT))
    stream = io.StringIO()
    result = unittest.TextTestRunner(stream=stream, verbosity=2).run(suite)
    log = stream.getvalue()
    (ROOT / 'reports/tests.log').write_text(log, encoding='utf-8')
    summary = {'tests_run':result.testsRun, 'failures':len(result.failures), 'errors':len(result.errors),
               'skipped':len(result.skipped), 'passed':result.testsRun-len(result.failures)-len(result.errors)-len(result.skipped),
               'successful':result.wasSuccessful()}
    (ROOT / 'reports/tests_summary.json').write_text(json.dumps(summary,indent=2)+'\n',encoding='utf-8')
    print(log)
    print(json.dumps(summary))
    return 0 if result.wasSuccessful() else 1


if __name__ == '__main__':
    sys.exit(main())
